cmake_minimum_required(VERSION 3.20)
project(jslice_trusted_verifier LANGUAGES CXX)
if(MSVC)
    # Parent preFlight uses /W3 and two conversion-warning suppressions. Scope
    # their removal to this directory so the independent target stays strict.
    string(REGEX REPLACE "/W[0-4]|/wd4244|/wd4267" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
endif()
get_filename_component(JS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(nlohmann_json 3.12 REQUIRED)
foreach(schema verifier-machine verification-manifest verification-work-packet)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${JS_ROOT}/docs/predictive-slicer/schema/${schema}.schema.json")
    file(READ "${JS_ROOT}/docs/predictive-slicer/schema/${schema}.schema.json" schema_text)
    string(REPLACE "-" "_" variable "${schema}")
    set(${variable} "${schema_text}")
endforeach()
configure_file(Schemas.hpp.in Schemas.hpp @ONLY)
add_library(jslice_trusted_verifier STATIC
    "${JS_ROOT}/src/TrustedVerifier/Verifier.cpp"
    "${JS_ROOT}/src/TrustedVerifier/Input.cpp")
target_compile_features(jslice_trusted_verifier PUBLIC cxx_std_20)
target_include_directories(jslice_trusted_verifier PUBLIC "${JS_ROOT}/src/TrustedVerifier" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(jslice_trusted_verifier PUBLIC nlohmann_json::nlohmann_json)
if(WIN32)
    target_link_libraries(jslice_trusted_verifier PRIVATE bcrypt)
else()
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(jslice_trusted_verifier PRIVATE OpenSSL::Crypto)
endif()
add_executable(jslice_verify "${JS_ROOT}/tools/trusted_verify.cpp")
target_link_libraries(jslice_verify PRIVATE jslice_trusted_verifier)
foreach(target jslice_trusted_verifier jslice_verify)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endforeach()
# An inspectable build contract: no slicer, IR, comparator or optimizer targets.
file(GENERATE OUTPUT "verifier-link-contract.txt" CONTENT
    "library=$<TARGET_PROPERTY:jslice_trusted_verifier,LINK_LIBRARIES>\nexecutable=$<TARGET_PROPERTY:jslice_verify,LINK_LIBRARIES>\n")
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(JS_VERIFIER_PRODUCTION_REPLAY "" CACHE PATH "Optional directory from a completed four-mode M1 production replay")
set(replay_args)
if(JS_VERIFIER_PRODUCTION_REPLAY)
    list(APPEND replay_args --production-replay "${JS_VERIFIER_PRODUCTION_REPLAY}")
endif()
add_test(NAME trusted_verifier_mutations COMMAND "${Python3_EXECUTABLE}"
    "${JS_ROOT}/tools/trusted_verifier_tests.py" "$<TARGET_FILE:jslice_verify>"
    --evidence "${CMAKE_CURRENT_BINARY_DIR}/verification-evidence.json" ${replay_args})
add_test(NAME m2_coupon_package COMMAND "${Python3_EXECUTABLE}"
    "${JS_ROOT}/tools/m2_coupon_tests.py")
set_tests_properties(m2_coupon_package PROPERTIES ENVIRONMENT "M2_COUPON_VERIFIER=$<TARGET_FILE:jslice_verify>")
